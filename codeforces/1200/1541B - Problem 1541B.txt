#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve()
{
    int n;
    cin >> n;

    vector<int> a(n + 1);
    vector<int> pos(2 * n + 1);

    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
        pos[a[i]] = i;
    }

    ll ans = 0;

    for (int i = 1; i <= n; i++)
    {
        // a[i] * a[j] = i + j <= 2*n
        for (int x = 1; x * a[i] <= 2 * n; x++)
        {
            if (pos[x] == 0)
            {
                continue;
            }

            int j = pos[x];

            if (i < j && a[i] * x == i + j)
            {
                ans++;
            }
        }
    }

    cout << ans << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--)
    {
        solve();
    }
}
