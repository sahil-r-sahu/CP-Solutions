#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve()
{
    ll n, x, y;
    cin >> n >> x >> y;

    ll common = n / lcm(x, y);
    ll plus = n / x - common;
    ll minus = n / y - common;

    // Sum of largest 'plus' numbers: n + (n-1) + ... + (n-plus+1)
    ll positiveSum = plus * (2 * n - plus + 1) / 2;

    // Sum of smallest 'minus' numbers: 1 + 2 + ... + minus
    ll negativeSum = minus * (minus + 1) / 2;

    cout << positiveSum - negativeSum << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--)
    {
        solve();
    }
}
