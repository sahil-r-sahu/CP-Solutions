#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--)
    {
        int n;
        cin >> n;

        int m = n * (n - 1) / 2;
        vector<long long> b(m);

        for (auto &x : b)
        {
            cin >> x;
        }

        sort(b.begin(), b.end());

        int idx = 0;

        for (int i = 0; i < n - 1; i++)
        {
            cout << b[idx] << " ";
            idx += n - i - 1;
        }
        cout << b.back() << '\n';
    }

    return 0;
}
