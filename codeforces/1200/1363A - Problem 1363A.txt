#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define pb push_back
#define endl '\n'

const int MOD = 1e9 + 7;
const int INF = 1e18;

void solve()
{
    int n, x;
    cin >> n >> x;

    vector<int> a(n);
    int od_cnt = 0, ev_cnt = 0;

    for (int i = 0; i < n; i++)
    {
        cin >> a[i];

        if (a[i] % 2 == 0)
        {
            ev_cnt++;
        }
        else
        {
            od_cnt++;
        }
    }

    for (int odd = 1; odd <= min(od_cnt, x); odd += 2)
    {
        int even = x - odd;

        if (even <= ev_cnt)
        {
            cout << "Yes" << endl;
            return;
        }
    }

    cout << "No" << endl;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(NULL);

    int t;
    cin >> t;

    while (t--)
    {
        solve();
    }
}
